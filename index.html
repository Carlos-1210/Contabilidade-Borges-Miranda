Configuração de Conexão

Encontrada em: Project Settings → API → Project URL

Encontrada em: Project Settings → API → Project API keys → "anon public" (NÃO use a "service_role")

Segurança

A chave anon do Supabase é pública por natureza (roda no navegador), mas isso só é seguro se você tiver Row Level Security (RLS) ativado e com políticas corretas nas tabelas clientes, recebimentos e pagamentos. Sem RLS, qualquer pessoa com essa chave lê e escreve todos os dados.

🗄️ Schema SQL das Tabelas

create table clientes (
  id uuid primary key default gen_random_uuid(),
  nome text not null,
  servico text,
  valor_mensal numeric not null default 0,
  dia_vencimento int not null default 1,
  status text not null default 'ativo', -- ativo | a_vencer | inadimplente
  criado_em timestamptz not null default now()
);

create table recebimentos (
  id uuid primary key default gen_random_uuid(),
  cliente_nome text not null,
  descricao text,
  valor numeric not null default 0,
  data date,
  vencimento date,
  status text not null default 'pendente', -- pendente | recebido | vencido
  criado_em timestamptz not null default now()
);

create table pagamentos (
  id uuid primary key default gen_random_uuid(),
  fornecedor text not null,
  descricao text,
  valor numeric not null default 0,
  data date,
  vencimento date,
  status text not null default 'pendente', -- pendente | pago
  criado_em timestamptz not null default now()
);

create table rotinas (
  id uuid primary key default gen_random_uuid(),
  tarefa text not null,
  alvo text,
  prazo date,
  status text not null default 'pendente', -- pendente | concluida
  criado_em timestamptz not null default now()
);

-- Ative RLS e crie policies adequadas antes de usar em produção:
alter table clientes enable row level security;
alter table recebimentos enable row level security;
alter table pagamentos enable row level security;
alter table rotinas enable row level security;

📋 Passo a Passo do Git

# Inicializar repositório
git init

# Arquivo .gitignore
echo "node_modules/" >> .gitignore
echo ".env" >> .gitignore

# Primeiro commit
git add .
git commit -m "Sistema Financeiro: versão inicial com Supabase"

# Conectar ao GitHub
git remote add origin https://github.com/seu-usuario/seu-repo.git
git branch -M main
git push -u origin main
